}

## 3. 复杂订阅操作

# 实时数据订阅
subscription UserActivityUpdates($userId: ID!) {
  userActivityUpdates(userId: $userId) {
    type
    timestamp
    data {
      ... on PostCreated {
        post {
          id
          title
          content
          author {
            name
            avatar
          }
        }
      }
      ... on CommentAdded {
        comment {
          id
          content
          author {
            name
          }
          post {
            id
            title
          }
        }
      }
      ... on LikeAdded {
        like {
          id
          user {
            name
          }
          post {
            id
            title
          }
        }
      }
    }
  }
}

## 4. 复杂片段查询

# 用户信息片段
fragment UserInfo on User {
  id
  name
  email
  avatar
  bio
  location {
    city
    country
  }
  socialLinks {
    platform
    url
  }
  createdAt
  updatedAt
}

# 文章信息片段
fragment PostInfo on Post {
  id
  title
  content
  excerpt
  author {
    ...UserInfo
  }
  tags {
    name
    slug
  }
  stats {
    likes
    comments
    shares
    views
  }
  createdAt
  updatedAt
}

# 评论信息片段
fragment CommentInfo on Comment {
  id
  content
  author {
    ...UserInfo
  }
  post {
    id
    title
  }
  parent {
    id
    content
  }
  replies {
    edges {
      node {
        ...CommentInfo
      }
    }
  }
  createdAt
  updatedAt
}

# 使用片段的复杂查询
query GetUserWithPostsAndComments($userId: ID!) {
  user(id: $userId) {
    ...UserInfo
    posts(first: 10) {
      edges {
        node {
          ...PostInfo
          comments(first: 5) {
            edges {
              node {
                ...CommentInfo
              }
            }
          }
        }
      }
    }
  }
}

## 6. 复杂指令查询

# 条件字段查询
query GetUserData($userId: ID!, $includePosts: Boolean!, $includeComments: Boolean!) {
  user(id: $userId) {
    id
    name
    email
    posts @include(if: $includePosts) {
      edges {
        node {
          id
          title
          content
          comments @include(if: $includeComments) {
            edges {
              node {
                id
                content
                author {
                  name
                }
              }
            }
          }
        }
      }
    }
  }
}
