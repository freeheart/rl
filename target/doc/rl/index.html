§使用示例

use rl::RL;
use std::path::Path;
 
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let rl = RL::new();
    let grammar = r#"
        grammar Calculator {
            expr: term (('+' | '-') term)*;
            term: factor (('*' | '/') factor)*;
            factor: number | '(' expr ')';
            number: [0-9]+;
        }
    "#;
 
    let rust_code = rl.compile_grammar_to_rust(grammar)?;
    rust_code.write_to_directory(Path::new("generated_parser"))?;
    Ok(())
}
